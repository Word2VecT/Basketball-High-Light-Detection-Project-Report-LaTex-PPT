\tikzset{

    input/.style={
        rectangle, draw=green!60!black, fill=green!5,

    },
    model/.style={
        rectangle, draw=blue!60!black, fill=blue!5,

        rounded corners, drop shadow, font=\sffamily\small
    },
    process/.style={
        rectangle, draw=orange!60!black, fill=orange!5,

    },
    decision/.style={
        diamond, draw=red!60!black, fill=red!5,

    },
    output/.style={
        rectangle, draw=purple!60!black, fill=purple!5, 
        text width=9em, text centered, minimum height=3.5em, 
        rounded corners, drop shadow, font=\sffamily\small
    },
    arrow/.style={
        thick, -{Stealth[scale=1.2]}, draw=gray!80

    label/.style={font=\sffamily\scriptsize\itshape, text=gray!80!black}

\begin{document}
\begin{tikzpicture}[node distance=1.5cm and 2cm]

    % --- Input Stage ---
    \node (detection) [input] {\textbf{输入数据 A}\\检测结果\\(BBox + ID)};
    \node (video) [input, below=of detection] {\textbf{输入数据 B}\\原始视频序列};
    
    \node[header, above=0.5cm of detection] {第一阶段: 数据输入};

    % --- Processing Stage ---
    \node (mmpose) [model, right=of $(detection.south)!0.5!(video.north)$] {\textbf{mmPose 模型}\\骨架姿态估计\\(支持 ID 追踪输入)};
    
    \draw [arrow] (detection.east) -- (mmpose.west);
    \draw [arrow] (video.east) -- (mmpose.west);

    % --- Skeleton Generation ---
    \node (skeletons) [process, right=of mmpose] {\textbf{骨架序列生成}\\多视角骨架数据\\(每帧/每 ID)};
    
    \draw [arrow] (mmpose) -- (skeletons);

    % --- Scoring & Selection ---
    \node (scoring) [process, below=of skeletons] {\textbf{评分机制}\\框置信度 +\\ 骨架完整度评分};
    
    \node (selection) [decision, left=of scoring] {\textbf{最优骨架选择}\\每帧选择\\最高分骨架};
    
    \draw [arrow] (skeletons) -- (scoring);
    \draw [arrow] (scoring) -- (selection);

    % --- Output Stage ---
    \node (visual) [output, below=of selection] {\textbf{最终可视化}\\高清骨架序列};
    
    \draw [arrow] (selection) -- (visual);

    % Annotations
    \node [label, right=0.2cm of skeletons, text width=4cm] {* 处理不同视角的\\骨架冗余};
    \node [label, left=0.2cm of visual, text width=4cm, align=right] {最终渲染展示 *};

    % Background Boxes
    \begin{scope}[on background layer]
        \node[fill=gray!5, draw=gray!30, dashed, rounded corners, fit=(detection) (video)] (bg_input) {};
        \node[fill=blue!2, draw=blue!20, dashed, rounded corners, fit=(mmpose) (skeletons)] (bg_model) {};
        \node[fill=red!2, draw=red!20, dashed, rounded corners, fit=(scoring) (selection)] (bg_select) {};
    \end{scope}
